def registration_constraint(sql):
    # Tokenize SQL without changing quoted values. This accepts the reviewed
    # migration grammar, not arbitrary equivalent SQL expressions.
    pattern=re.compile(r"\s+|'(?:''|[^'])*'|\"(?:\"\"|[^\"])*\"|[A-Za-z_][A-Za-z_0-9]*|[0-9]+|[(),;=<>+*/.-]")
    tokens=[];pos=0
    while pos<len(sql):
        # Quoted tokens are consumed whole below, so this guard only sees
        # unquoted SQL. Reject comments, including unterminated block comments.
        require(not sql.startswith(('--','/*'),pos),'SQL comments are not permitted in registration DDL')
        match=pattern.match(sql,pos)
        require(match is not None,'Unrecognized registration DDL token')
        value=match.group();pos=match.end()
        if value.isspace():continue
        if value.startswith("'"):tokens.append(('literal',value[1:-1].replace("''", "'")))
        elif value.startswith('"'):tokens.append(('identifier',value[1:-1].replace('""','"')))

    expected=[('sql',v) for v in ('constraint','ck_ledger_operation_state','check','(','state','in','(')]

### Recovery state table

| Durable state | Meaning and permitted response |
|---|---|
| CREATED | Command saved, no accepted durable preparation. Inspect failure; separately authorize prepare-created only. No submit. |
| PREPARED | Artifact durable, submission gate not crossed. Inspect/restart operator safely; one new D authorization required. |
| SUBMISSION_PENDING | CAS durable; may have crashed before or during network call. Treat outcome as ambiguous; status-only reconciliation. |
| SUBMITTED | Acknowledged, not committed. Reconcile only. |
| COMMITTED_VALID | Confirmed code 0. Check receipts/property and CP evidence; never submit again. |
| COMMITTED_INVALID | Terminal invalidity. Preserve code/block; stop and investigate. No retry/new property operation. |
| UNKNOWN | Submission/status uncertain. Preserve all state; reconcile only, no automatic resubmission. |
